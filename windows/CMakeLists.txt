cmake_minimum_required(VERSION 3.16)
project(mb86H_windows C)
if(NOT WIN32 OR NOT MINGW OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use x86_64-w64-mingw32 (windows/shell.nix + build.sh)")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CORE ${CMAKE_CURRENT_SOURCE_DIR}/..)
add_library(xz_winusb STATIC winusb/transport.c winusb/host.c winusb/binary_file.c
    ${CORE}/core/loader.c ${CORE}/core/bootstrap.c ${CORE}/core/recovery.c ${CORE}/core/resident_image.c
    ${CORE}/core/session.c ${CORE}/core/device_query.c ${CORE}/core/frontend.c ${CORE}/core/protocol.c
    ${CORE}/core/receiver.c ${CORE}/core/ts_framer.c ${CORE}/core/ts_mux.c
    ${CORE}/usb/window.c ${CORE}/device/gv_mvp_xz/device.c ${CORE}/device/gv_mvp_xz/boot.c)
target_include_directories(xz_winusb PUBLIC ${CORE}/include winusb)
target_compile_definitions(xz_winusb PUBLIC _WIN32_WINNT=0x0601 UNICODE _UNICODE)
target_compile_options(xz_winusb PRIVATE -Wall -Wextra -Werror)
target_link_libraries(xz_winusb PUBLIC winusb setupapi uuid advapi32 ole32)
add_library(xz_ipc STATIC winusb/ipc.c winusb/client.c)
target_include_directories(xz_ipc PUBLIC winusb ${CORE}/include)
target_compile_definitions(xz_ipc PUBLIC _WIN32_WINNT=0x0601 UNICODE _UNICODE)
target_compile_options(xz_ipc PRIVATE -Wall -Wextra -Werror)
target_link_libraries(xz_ipc PUBLIC advapi32 user32)
add_executable(DriverHost_mb86H winusb/driver_host.c winusb/ipc.c)
target_link_libraries(DriverHost_mb86H PRIVATE xz_winusb user32)
target_compile_options(DriverHost_mb86H PRIVATE -Wall -Wextra -Werror)
target_link_options(DriverHost_mb86H PRIVATE -municode)
add_library(BonDriver_mb86H SHARED winusb/bondriver.c winusb/bon_abi.c)
set_target_properties(BonDriver_mb86H PROPERTIES PREFIX "")
target_sources(BonDriver_mb86H PRIVATE ${CORE}/core/frontend.c)
target_link_libraries(BonDriver_mb86H PRIVATE xz_ipc)
target_compile_options(BonDriver_mb86H PRIVATE -Wall -Wextra -Werror)
add_executable(xz-capture winusb/capture.c)
target_compile_options(xz-capture PRIVATE -Wall -Wextra -Werror)
target_sources(xz-capture PRIVATE ${CORE}/core/frontend.c winusb/binary_file.c)
target_link_libraries(xz-capture PRIVATE xz_ipc)
target_link_options(xz-capture PRIVATE -municode)
set(MSVC_CONSUMER "${CMAKE_CURRENT_BINARY_DIR}/msvc_consumer.obj")
set_source_files_properties(${MSVC_CONSUMER} PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
add_executable(bon-abi-probe winusb/abi_probe.c ${MSVC_CONSUMER})
target_include_directories(bon-abi-probe PRIVATE winusb ${CORE}/include)
target_compile_options(bon-abi-probe PRIVATE -Wall -Wextra -Werror)
add_executable(host-ipc-probe tests/host_ipc_probe.c winusb/driver_host.c winusb/ipc.c winusb/client.c)
target_include_directories(host-ipc-probe PRIVATE winusb ${CORE}/include)
target_compile_definitions(host-ipc-probe PRIVATE _WIN32_WINNT=0x0601 UNICODE _UNICODE XZ_IPC_TEST)
target_compile_options(host-ipc-probe PRIVATE -Wall -Wextra -Werror)
target_sources(host-ipc-probe PRIVATE ${CORE}/core/frontend.c)
target_link_libraries(host-ipc-probe PRIVATE advapi32 user32)
target_link_options(host-ipc-probe PRIVATE -municode)
foreach(t BonDriver_mb86H xz-capture bon-abi-probe DriverHost_mb86H host-ipc-probe)
    target_link_options(${t} PRIVATE -static -static-libgcc)
    if(DEFINED ENV{MCFGTHREAD_LIB_DIR})
        target_link_directories(${t} PRIVATE "$ENV{MCFGTHREAD_LIB_DIR}")
    endif()
endforeach()
